cmake_minimum_required(VERSION 3.20)
project(npxf VERSION 1.0.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME MATCHES "^(Linux|Darwin)$")
  message(FATAL_ERROR "npxf supports Linux and macOS")
endif()
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
include(CTest)
include(GNUInstallDirs)
set(NPXF_SANITIZER "" CACHE STRING "Sanitizer: address or thread")
add_library(npxf_common STATIC src/crypto.cpp src/crypto_vectors.cpp src/net.cpp src/channel.cpp src/ninep.cpp)
target_compile_features(npxf_common PUBLIC cxx_std_20)
target_link_libraries(npxf_common PUBLIC OpenSSL::Crypto Threads::Threads)
target_compile_options(npxf_common PUBLIC -Wall -Wextra -Wpedantic -fstack-protector-strong)
if(NPXF_SANITIZER)
  if(NPXF_SANITIZER STREQUAL "address")
    set(sanitize "address,undefined")
  elseif(NPXF_SANITIZER STREQUAL "thread")
    set(sanitize "thread")
  else()
    message(FATAL_ERROR "Unknown NPXF_SANITIZER")
  endif()
  target_compile_options(npxf_common PUBLIC -fsanitize=${sanitize} -fno-omit-frame-pointer)
  target_link_options(npxf_common PUBLIC -fsanitize=${sanitize})
endif()
add_executable(npxf src/client.cpp src/npxf_main.cpp)
add_executable(npxf-server src/server.cpp src/server_ops.cpp)
target_link_libraries(npxf PRIVATE npxf_common)
target_link_libraries(npxf-server PRIVATE npxf_common)
if(BUILD_TESTING)
  add_executable(npxf-selftest src/selftest.cpp)
  target_link_libraries(npxf-selftest PRIVATE npxf_common)
  add_test(NAME npxf-selftest COMMAND npxf-selftest)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  find_program(BASH_EXECUTABLE bash REQUIRED)
  add_test(NAME npxf-integration COMMAND ${BASH_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test.sh)
  set_tests_properties(npxf-selftest npxf-integration PROPERTIES TIMEOUT 180)
  set_tests_properties(npxf-integration PROPERTIES
    ENVIRONMENT "NPXF_TEST_BIN_DIR=$<TARGET_FILE_DIR:npxf-server>")
endif()
install(TARGETS npxf npxf-server RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
